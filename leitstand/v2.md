```bash
sudo mkdir -p "$BASE_DIR"
sudo chown -R "$USER":"$USER" "$BASE_DIR"

###############################################################################
# (1) leitstand.config.json (dump-konform)
###############################################################################
cat > "$BASE_DIR/leitstand.config.json" <<'JSON'
{
  "paths": {
    "semantah": {
      "todayInsights": "$VAULT_ROOT/.gewebe/insights/today.json"
    },
    "chronik": {
      "dataDir": "$VAULT_ROOT/.gewebe/chronik/data"
    },
    "wgx": {
      "metricsDir": "$VAULT_ROOT/.gewebe/wgx/metrics"
    }
  },
  "output": {
    "dir": "digests/daily"
  },
  "digest": {
    "maxEvents": 20
  }
}
JSON

###############################################################################
# (2) Apply-Script: DOCKER-USER cage für 80/443 (LAN+WG only)
###############################################################################
sudo tee /usr/local/sbin/heimgewebe-gateway-firewall-apply >/dev/null <<'SH'
#!/usr/bin/env bash
set -euo pipefail

# Kanonische Vertrauenszonen (heimserver.context.md):
LAN_CIDR="192.168.178.0/24"
WG_CIDR="10.7.0.0/24"

# Ports, die als "Entry-Gateway" gelten (Caddy in Docker)
PORTS="80,443"

# Safety: DOCKER-USER muss existieren (Docker legt sie normalerweise an).
# Falls Docker noch nicht läuft, bricht das Script sauber ab.
if ! sudo iptables -nL DOCKER-USER >/dev/null 2>&1; then
  echo "DOCKER-USER chain not found (Docker not running yet?)."
  echo "Start Docker first, then rerun."
  exit 1
fi

ensure_rule() {
  local rule=("$@")
  if sudo iptables -C "${rule[@]}" 2>/dev/null; then
    return 0
  fi
  sudo iptables -I "${rule[@]}"
}

# 1) Allow 80/443 from LAN
ensure_rule DOCKER-USER -p tcp -m multiport --dports "$PORTS" -s "$LAN_CIDR" -j ACCEPT
# 2) Allow 80/443 from WireGuard
ensure_rule DOCKER-USER -p tcp -m multiport --dports "$PORTS" -s "$WG_CIDR" -j ACCEPT

# 3) Drop 80/443 from everywhere else
ensure_rule DOCKER-USER -p tcp -m multiport --dports "$PORTS" -j DROP

# 4) Ensure RETURN exists (non-relevant traffic continues)
# (If it already exists, no-op)
if ! sudo iptables -C DOCKER-USER -j RETURN 2>/dev/null; then
  sudo iptables -A DOCKER-USER -j RETURN
fi

echo "Applied DOCKER-USER cage rules for ports ${PORTS} (allow: ${LAN_CIDR}, ${WG_CIDR}; drop others)."
SH
sudo chmod +x /usr/local/sbin/heimgewebe-gateway-firewall-apply

###############################################################################
# (3) systemd service: apply rules at boot (persisted via netfilter-persistent)
###############################################################################
sudo tee /etc/systemd/system/heimgewebe-gateway-firewall.service >/dev/null <<'UNIT'
[Unit]
Description=Heimgewebe Gateway Firewall Cage (DOCKER-USER 80/443 LAN+WG only)
Wants=docker.service
After=docker.service

[Service]
Type=oneshot
ExecStart=/usr/local/sbin/heimgewebe-gateway-firewall-apply

# Optional: After applying, persist rules if netfilter-persistent is present.
# This keeps the "auditfest" property described in the context.
ExecStartPost=/bin/sh -c 'command -v netfilter-persistent >/dev/null 2>&1 && netfilter-persistent save || true'

RemainAfterExit=yes

[Install]
WantedBy=multi-user.target
UNIT

###############################################################################
# (4) enable + run now
###############################################################################
sudo systemctl daemon-reload
sudo systemctl enable --now heimgewebe-gateway-firewall.service

echo
echo "== Verify =="
sudo iptables -S DOCKER-USER | sed -n '1,120p'
```
